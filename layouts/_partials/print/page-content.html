{{- /* OINK v1.5.1, with literal-code guards added to the returned HTML. */ -}} {{- /* One cache key owns both Print render modes for a Book page. Hugo may render the plain page and overlapping Book aggregates in parallel, so computing the variants in a fixed order keeps tdBookAggregate local to this coordinator instead of racing two differently keyed partials. */ -}} {{- $page := .page -}} {{- $page.Store.Set "tdOutputFormat" "print" -}} {{- $page.Store.Delete "tdBookAggregate" -}} {{- $plain := $page.RenderString (dict "display" "block") $page.RawContent -}} {{- $plainHeadings := partial "print/heading-ids.html" $plain -}} {{- $plain = partial "content/static-image-output.html" $plain -}} {{- partial "code/register-rendered-ids.html" (dict "page" $page "content" $plain) -}} {{- if eq $page.Type "expo" }}{{ $plain = printf "%s%s" $plain (partial "expo/static.html" $page) | safeHTML }}{{ end -}} {{- $plain = partial "print/resolve-links.html" (dict "page" $page "content" $plain) -}} {{- $book := $plain -}} {{- $bookHeadings := $plainHeadings -}} {{- if eq $page.Type "book" -}} {{- $page.Store.Set "tdBookAggregate" true -}} {{- $book = $page.RenderString (dict "display" "block") $page.RawContent -}} {{- $bookHeadings = partial "print/heading-ids.html" $book -}} {{- $book = partial "content/static-image-output.html" $book -}} {{- partial "code/register-rendered-ids.html" (dict "page" $page "content" $book) -}} {{- $book = partial "print/resolve-links.html" (dict "page" $page "content" $book) -}} {{- $page.Store.Delete "tdBookAggregate" -}} {{- end -}} {{- $plain = partial "content/literal-code.html" $plain -}} {{- $book = partial "content/literal-code.html" $book -}} {{- return (dict "plain" $plain "book" $book "plainHeadings" $plainHeadings "bookHeadings" $bookHeadings) -}}
